# tasteful-todo M1 (3-Column UI) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** mock 데이터로 동작하는 Tauri + React 3-column UI — 코어 루프(goal 선택 → task 작성 → 타임라인 드래그 배치 → 체크)와 역방향 생성(타임라인 드래그 → 팝오버)까지 전부 인터랙션 가능한 상태.

**Architecture:** zustand 스토어가 in-memory 도메인 상태(Goal/Task/TimeBlock)를 소유하고, 세 컬럼 컴포넌트는 스토어만 바라본다. 시간↔픽셀 변환과 스냅은 순수 함수 라이브러리(`lib/time.ts`)로 분리해 TDD. 파일 I/O(M2)·EventKit(M3)은 이 플랜 범위 밖 — 캘린더 일정도 mock이다. M1의 에디터는 구조화된 task 리스트 UI이며, 자유 markdown 편집(CodeMirror 6)은 M2에서 파싱 엔진과 함께 도입한다.

**Tech Stack:** Tauri 2.x, React 19, TypeScript 5, Vite 7, zustand 5, @dnd-kit/core 6, Vitest 3 + React Testing Library, @fontsource (Hahmlet / IBM Plex Sans KR / IBM Plex Mono)

**Spec:** `docs/specs/2026-08-23-tasteful-todo-design.md` (§8 UI 디자인이 시각 기준, 목업 `docs/mockups/tasteful-todo-ui.html`과 대조)

## Global Constraints

- M1은 파일 I/O·EventKit 금지 — 모든 데이터는 `src/mock/data.ts`에서만 온다 (spec §6 M1).
- 디자인 토큰은 spec §8 표의 hex 값을 그대로 사용. 다크 테마는 `prefers-color-scheme` 미디어 쿼리로 (spec §8).
- 시간 격자: 48px/시간, 15분 스냅, 최소 블록 15분, 하루 범위 00:00–24:00 (초기 스크롤 08:00).
- 컬럼 폭: 240px / flex / 300px. 창 최소 폭 1080px.
- UI 문구는 한국어. 커밋 메시지는 conventional commits (`feat:`/`test:`/`chore:`).
- task 상태 변경은 반드시 스토어 액션을 통해서만 — 컴포넌트가 배열을 직접 만지지 않는다.
- 각 태스크 종료 시 `pnpm exec vitest run`이 전부 통과해야 커밋한다.

```ts
import { create } from 'zustand';
import type { CalendarEvent, Goal, GoalId, Task, TimeBlock } from './types';

export function todayISO(): string {
  const d = new Date();
  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`;
}

export function shiftDate(date: string, days: number): string {
  const [y, m, dd] = date.split('-').map(Number);
  const next = new Date(y, m - 1, dd + days);
  return `${next.getFullYear()}-${String(next.getMonth() + 1).padStart(2, '0')}-${String(next.getDate()).padStart(2, '0')}`;
}

export interface AppState {
  goals: Goal[];
  tasks: Task[];
  blocks: TimeBlock[];
  calendarEvents: CalendarEvent[];
  selectedGoalId: GoalId;
  currentDate: string;
  selectGoal: (id: GoalId) => void;
  goPrevDay: () => void;
  goNextDay: () => void;
  goToday: () => void;
  toggleTask: (taskId: string) => void;
  addTask: (goalId: GoalId, text: string) => string;
  updateTaskText: (taskId: string, text: string) => void;
}

// mock import는 순환을 피하기 위해 함수 아래에서
import { mockBlocks, mockCalendarEvents, mockGoals, mockTasks } from './mock/data';

export const useApp = create<AppState>((set) => ({
  goals: mockGoals,
  tasks: mockTasks,
  blocks: mockBlocks,
  calendarEvents: mockCalendarEvents,
  selectedGoalId: 'side-project',
  currentDate: todayISO(),

  selectGoal: (id) => set({ selectedGoalId: id }),
  goPrevDay: () => set((s) => ({ currentDate: shiftDate(s.currentDate, -1) })),
  goNextDay: () => set((s) => ({ currentDate: shiftDate(s.currentDate, 1) })),
  goToday: () => set({ currentDate: todayISO() }),

  toggleTask: (taskId) =>
    set((s) => ({ tasks: s.tasks.map((t) => (t.id === taskId ? { ...t, done: !t.done } : t)) })),

  addTask: (goalId, text) => {
    const id = crypto.randomUUID();
    set((s) => ({ tasks: [...s.tasks, { id, goalId, text, done: false }] }));
    return id;
  },

  updateTaskText: (taskId, text) =>
    set((s) => ({ tasks: s.tasks.map((t) => (t.id === taskId ? { ...t, text } : t)) })),
}));

export const openCount = (s: AppState, goalId: GoalId) =>
  s.tasks.filter((t) => t.goalId === goalId && !t.done).length;

export const blocksForDate = (s: AppState, date: string) =>
  s.blocks.filter((b) => b.date === date);

export const plannedMinutes = (s: AppState, date: string) =>
  blocksForDate(s, date).reduce((sum, b) => sum + (b.endMin - b.startMin), 0);
```

```ts
export const PX_PER_HOUR = 48;
export const SNAP_MIN = 15;
export const DAY_MIN = 1440;

export const minToY = (min: number) => (min / 60) * PX_PER_HOUR;
export const yToMin = (y: number) => (y / PX_PER_HOUR) * 60;

export const snap = (min: number) =>
  Math.max(0, Math.min(DAY_MIN, Math.round(min / SNAP_MIN) * SNAP_MIN));

export function clampRange(start: number, end: number): [number, number] {
  let s = Math.max(0, start);
  let e = Math.min(DAY_MIN, Math.max(end, s + SNAP_MIN));
  if (e > DAY_MIN) e = DAY_MIN;
  if (e - s < SNAP_MIN) s = e - SNAP_MIN;
  return [s, e];
}

export const fmtTime = (min: number) =>
  `${String(Math.floor(min / 60)).padStart(2, '0')}:${String(min % 60).padStart(2, '0')}`;

export const fmtRange = (start: number, end: number) => `${fmtTime(start)}–${fmtTime(end)}`;

export function fmtPlanned(totalMin: number): string {
  const h = Math.floor(totalMin / 60);
  const m = totalMin % 60;
  if (h === 0) return `${m}m`;
  return m === 0 ? `${h}h` : `${h}h ${m}m`;
}

export const newBlockRefId = () =>
  Array.from({ length: 4 }, () => '0123456789abcdefghijklmnopqrstuvwxyz'[Math.floor(Math.random() * 36)]).join('');
```

```ts
createBlock: (startMin: number, endMin: number, title: string, goalId?: GoalId) => string;

scheduleTask: (taskId, startMin, durMin = 60) =>
  set((s) => {
    const task = s.tasks.find((t) => t.id === taskId);
    if (!task) return s;
    const [start, end] = clampRange(snap(startMin), snap(startMin) + durMin);
    const blockRefId = task.blockRefId ?? newBlockRefId();
    return {
      tasks: s.tasks.map((t) => (t.id === taskId ? { ...t, blockRefId } : t)),
      blocks: [...s.blocks, { id: crypto.randomUUID(), date: s.currentDate, startMin: start, endMin: end, taskId }],
    };
  }),

createBlock: (startMin, endMin, title, goalId) => {
  const blockId = crypto.randomUUID();
  set((s) => {
    const [start, end] = clampRange(snap(startMin), snap(endMin));
    if (!goalId) {
      return { blocks: [...s.blocks, { id: blockId, date: s.currentDate, startMin: start, endMin: end, title }] };
    }
    const task = { id: crypto.randomUUID(), goalId, text: title, done: false, blockRefId: newBlockRefId() };
    return {
      tasks: [...s.tasks, task],
      blocks: [...s.blocks, { id: blockId, date: s.currentDate, startMin: start, endMin: end, taskId: task.id }],
    };
  });
  return blockId;
},

moveBlock: (blockId, newStartMin) =>
  set((s) => ({
    blocks: s.blocks.map((b) => {
      if (b.id !== blockId) return b;
      const dur = b.endMin - b.startMin;
      let start = snap(newStartMin);
      if (start + dur > DAY_MIN) start = DAY_MIN - dur;
      if (start < 0) start = 0;
      return { ...b, startMin: start, endMin: start + dur };
    }),
  })),

resizeBlock: (blockId, newEndMin) =>
  set((s) => ({
    blocks: s.blocks.map((b) =>
      b.id === blockId
        ? { ...b, endMin: Math.min(DAY_MIN, Math.max(b.startMin + SNAP_MIN, snap(newEndMin))) }
        : b,
    ),
  })),

deleteBlock: (blockId) =>
  set((s) => ({ blocks: s.blocks.filter((b) => b.id !== blockId) })),
```

```tsx
          {HOURS.map((h) => (

function NowLine({ date }: { date: string }) {

  return <div className="nowline" style={{ top: minToY(now.getHours() * 60 + now.getMinutes()) }} />;
}
```
